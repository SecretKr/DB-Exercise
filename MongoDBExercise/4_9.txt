db.orders.aggregate([
  {$lookup: {from: "customer", localField: "customer_id", foreignField: "customer_id", as: "combine"}},
  {$unwind: "$combine"},
  {
      $match: {
          order_date: {
              $gte: "2020-01-10 00:00:00",
              $lte: "2020-01-15 23:59:59"
          }
      }
  },
  {$group: {"_id": "$customer_id", customer_id: {$first: "$customer_id"}, customer_name: {$first: "$combine.customer_name"}}},
  {$sort: { customer_id: 1 }},
  {$project: { _id: 0, customer_id: 1, customer_name: 1}},
])
